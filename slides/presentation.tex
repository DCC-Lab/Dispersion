\begin{frame}{Motivation: Imaging Myelin with Light}
  \begin{columns}[T]
    \column{0.55\textwidth}
      \textbf{Goal:} chemically-selective, label-free imaging of myelin sheaths

        \item No labels required

      \end{tikzpicture}

    \column{0.48\textwidth}
      \textbf{Phase-matching condition:}
      \[
        \omega_{\rm AS} = 2\omega_p - \omega_S
      \]

      \vspace{0.5em}
      \textbf{Resonance condition:}
      \[
        \Omega = \omega_p - \omega_S \approx \omega_{\rm Raman}
      \]

      \vspace{0.5em}
      \textbf{Signal scales as:}
      \[
        I_{\rm CARS} \propto |A_{\rm Raman}(\Omega)|^2 \cdot I_p^2 \cdot I_S
      \]

      \vspace{0.6em}
      \begin{itemize}
        \item Anti-Stokes \textcolor{mgreen}{$\omega_{\rm AS}$} is blue-shifted\\
              away from excitation $\Rightarrow$ easy to detect
        \item Here: pump at \SI{803}{nm}, Stokes at \SI{1041}{nm}\\
              $\Rightarrow$ anti-Stokes $\approx$ \SI{654}{nm} ($\approx 15\,292\wn$)
      \end{itemize}
  \end{columns}
\end{frame}

% ── Slide 4: Spectral Focusing Concept ───────────────────────
\begin{frame}{Spectral Focusing: Chirp as a Spectral Filter}
  \begin{columns}[T]
    \column{0.5\textwidth}
      \textbf{Without chirp (transform-limited):}
      \begin{itemize}
        \item Both beams are broad in frequency
        \item All frequency pairs $(\omega_p, \omega_S)$ overlap\\
              simultaneously
        \item Many Raman modes excited at once\\
              $\Rightarrow$ \alert{poor spectral resolution}
      \end{itemize}

      \vspace{0.8em}
      \textbf{With matched chirp (spectral focusing):}
      \begin{itemize}
        \item Glass disperses each pulse: each colour arrives\\
              at a different time
        \item At any instant $t$, only one narrow frequency\\
              pair $(\omega_p(t),\,\omega_S(t))$ is active
        \item Instantaneous beat frequency \emph{constant}:\\
              $\omega_p(t) - \omega_S(t) \approx \Omega_0$
        \item $\Rightarrow$ \textcolor{mgreen}{\textbf{single Raman mode addressed per time slice}}
      \end{itemize}

    \column{0.47\textwidth}
      \begin{tikzpicture}[scale=0.8]

% ── Slide 6: S-TIH6 Glass Dispersion ────────────────────────
\begin{frame}{S-TIH6 Glass: Dispersion and Chirp Direction}
  \begin{columns}[T]
    \column{0.52\textwidth}
      \textbf{Sellmeier equation} for $n(\lambda)$:
      \[
        n^2(\lambda) = 1 + \sum_{j=1}^{3}\frac{B_j\,\lambda^2}{\lambda^2 - C_j}
      \]

      \vspace{0.5em}
      \textbf{Group index:}
      \[
        n_g(\lambda) = n(\lambda) - \lambda\,\frac{dn}{d\lambda}
      \]

      Since $\dfrac{dn}{d\lambda} < 0$ (normal dispersion):
      $\quad n_g > n$

      \vspace{0.5em}
      \textbf{Relative group delay} of spectral slice $\lambda_i$:
      \[
        \tau_g(\lambda_i) = \frac{d\bigl[n_g(\lambda_i) - n_g(\lambda_0)\bigr]}{c}
      \]

      \vspace{0.4em}
      \begin{itemize}
        \item $\lambda_i < \lambda_0$ (blue): $n_g$ larger $\Rightarrow$ \textbf{arrives later} ($\tau_g > 0$)
        \item $\lambda_i > \lambda_0$ (red): $n_g$ smaller $\Rightarrow$ \textbf{arrives earlier} ($\tau_g < 0$)
      \end{itemize}

    \column{0.45\textwidth}
      \begin{alertblock}{Chirp convention}
        \textbf{Time domain} (at the sample):\\[0.3em]
        \quad Red arrives first $\to$ Blue arrives last\\[0.3em]
        \quad $\omega(t)$ increases with time\\[0.3em]
        \quad $\Rightarrow$ \textbf{positive chirp}

        \vspace{0.5em}
        \textbf{Spatial snapshot} (pulse in flight):\\[0.3em]
        \quad Red at the front (faster)\\[0.3em]
        \quad $\Rightarrow$ "negative chirp" by spatial convention

        \vspace{0.3em}
        {\small Same physics, two viewpoints.\\
        This simulation uses the \emph{time-domain} convention.}
      \end{alertblock}

      \vspace{0.4em}
      \begin{block}{Group-velocity dispersion}
        GVD $> 0$ for S-TIH6 at 800–1050 nm.\\
        Normal dispersion $\Rightarrow$ pulse stretching.
      \end{block}
  \end{columns}
\end{frame}

\section{Pulse Chirping Through Glass}

% ── Slide 8: Field vs Intensity Convention ───────────────────
\begin{frame}{Physical Convention: Field vs.\ Intensity}
  \begin{columns}[T]
    \column{0.5\textwidth}
      \textbf{Pulse stored as field amplitude $E$, not intensity $I$:}
      \[
        E(t, \lambda_i) = A_E(\lambda_i)\,\exp\!\left(-\frac{(t - \tau_g(\lambda_i))^2}{\taue^2}\right)
      \]

      \begin{itemize}
        \item $A_E(\lambda_i)$: spectral weight (field Gaussian envelope)
        \item $\taue$: field Gaussian parameter
        \item $\tau_g(\lambda_i)$: relative group delay of slice $\lambda_i$
        \item Intensity plotted as $I = E^2$
      \end{itemize}

      \vspace{0.5em}
      \textbf{Field vs.\ intensity FWHM:}
      \[
        \mathrm{FWHM}_I = \frac{\mathrm{FWHM}_E}{\sqrt{2}}
      \]

    \column{0.47\textwidth}
      \textbf{Why use field amplitudes for convolutions?}

      \vspace{0.4em}
      Raman coherence is driven by:
      \[
        \rho(\Omega, t) \propto \Epump(t)\cdot\Estokes^*(t)
      \]

      This is a \emph{field $\times$ field} product — not $\sqrt{I}\times\sqrt{I}$.

      \vspace{0.5em}
      \begin{alertblock}{Using intensities would be wrong}
        $I_p \cdot I_S = E_p^2 \cdot E_S^2$ overestimates the\\
        effective nonlinear bandwidth by $\sim\!\sqrt{2}$,\\
        artificially narrowing the Raman linewidth.
      \end{alertblock}

      \vspace{0.4em}
      \textbf{Summary:}
      \begin{tabular}{ll}
        \toprule
        Quantity & Level \\
        \midrule
        $E_{\rm init}$, $E_{\rm prop}$ & Field \\
        3D pulse plots & Intensity ($E^2$) \\
        $C_1$, $C_2$ & Field amplitude \\
        \bottomrule
      \end{tabular}
  \end{columns}
\end{frame}

% ── Slide 11: Peak Power Reduction — Pump ────────────────────
\begin{frame}{Peak Power Reduction: The Musicians Analogy}
  \begin{columns}[T]
    \column{0.55\textwidth}
      \begin{figure}
        \includegraphics[width=\linewidth, keepaspectratio]
          {../plots/resolution/pump_step1b_marginal.pdf}
        \caption*{\small Marginal temporal intensity $I(t) = \sum_i E^2(\lambda_i, t)$}
      \end{figure}

    \column{0.43\textwidth}
      \textbf{Why does the peak drop?}

      Glass is \emph{linear and lossless} — each spectral slice
      $A_k$ is unchanged. But:

      \vspace{0.4em}
      \begin{tabular}{p{3.8cm}l}
        \textbf{TL pulse:} all $n_{\rm spec}$ slices peak \emph{together} at $t=0$
        & $\sum_i A_i^2$ \\[0.4em]
        \textbf{Chirped pulse:} slice $k$ peaks alone at $t = \tau_g(k)$; others negligible
        & $A_k^2$ \\
      \end{tabular}

      \vspace{0.5em}
      \[
        \frac{I_{\rm peak}^{\rm chirped}}{I_{\rm peak}^{\rm TL}}
          = \frac{\max_k A_k^2}{\sum_i A_i^2}
          = \frac{1}{\neff}
      \]

      \textbf{Measured:} ratio $= 0.080$, so $\neff \approx 12$

      \vspace{0.4em}
      \begin{alertblock}{Energy is conserved}
        $\int I(t)\,dt$ identical before and after glass. \\
        Measured: ratio $= 1.000\;(\checkmark)$
      \end{alertblock}
  \end{columns}
\end{frame}

\section{CARS Signal Chain}

% ── Slide 14: C1 Formula ─────────────────────────────────────
\begin{frame}{Raman Excitation — $C_1(\Omega, t)$}
  \begin{columns}[T]
    \column{0.52\textwidth}
      The \textbf{Raman coherence amplitude} driven by the chirped pulse pair:

      \vspace{0.5em}
      \[
        \boxed{
          C_1(\Omega, t) = \sum_k \Epump(\nu_k, t)\cdot\Estokes(\nu_k - \Omega,\, t)
        }
      \]

      \vspace{0.5em}
      \begin{itemize}
        \item $\Omega$: Raman shift (scanned over $2500$–$3800\wn$)
        \item $\nu_k$: pump frequency at spectral slice $k$
        \item $\nu_k - \Omega$: matching Stokes frequency (CARS condition)
        \item Sum over all pump spectral slices
      \end{itemize}

      \vspace{0.5em}
      \textbf{Physical meaning:} at each time $t$, only
      those pump–Stokes pairs that are simultaneously present
      contribute — this is the spectral-focusing mechanism.

    \column{0.45\textwidth}
      \begin{block}{Implementation}
        \begin{itemize}
          \item Stokes interpolated onto a fine\\
                wavenumber grid
          \item Index arithmetic maps pump $\nu_k$\\
                to Stokes $\nu_k - \Omega$
          \item Vectorised: $\sim\SI{10}{ms}$ per call
          \item Scanned over $\delta t \in [-8, +8]\,\text{ps}$\\
                (150 points) to find optimal delay
        \end{itemize}
      \end{block}

      \vspace{0.4em}
      \begin{block}{Output}
        2D array $C_1[\Omega, t]$:\\[0.3em]
        \quad $N_\Omega = 300$ (Raman axis)\\
        \quad $N_t = 1000$ (time axis)
      \end{block}
  \end{columns}
\end{frame}

% ── Slide 16: Optimal Delay ──────────────────────────────────
\begin{frame}{Finding the Optimal Pump–Stokes Delay}
  \begin{columns}[T]
    \column{0.52\textwidth}
      \textbf{Problem:} when chirp rates are mismatched (Scenario B),
      the centres of mass of the two chirped pulses do not coincide.

      \vspace{0.6em}
      \textbf{Criterion — maximise total $C_1$ signal:}
      \[
        \delta t^\star = \arg\max_{\delta t}\;\sum_{\Omega, t} C_1(\Omega, t;\,\delta t)
      \]

      This maximises \emph{temporal overlap}, not spectral resolution.

      \vspace{0.6em}
      \begin{itemize}
        \item \textbf{Scenario A} (matched, 150/150 mm):\\
              $\delta t^\star \approx 0$ — pulses naturally aligned
        \item \textbf{Scenario B} (mismatched, 50/150 mm):\\
              $\delta t^\star \neq 0$ — pump less stretched, must shift
      \end{itemize}

    \column{0.45\textwidth}
      \begin{alertblock}{What optimal delay is \emph{not}}
        It is \textbf{not} the delay that minimises\\
        the spectral width of $C_1(\Omega)$.

        For mismatched chirp rates, the signal-maximising
        and resolution-optimising delays are different.
      \end{alertblock}

      \vspace{0.6em}
      \begin{block}{Algorithm}
        Brute-force scan over 150 delay values\\
        $\delta t \in [-8,\,+8]\,\text{ps}$\\
        $\approx \SI{0.6}{s}$ total
      \end{block}
  \end{columns}
\end{frame}

% ── Slide 18: C2 Formula ─────────────────────────────────────
\begin{frame}{Anti-Stokes Signal — $C_2(\nu_{\rm AS}, t)$}
  \begin{columns}[T]
    \column{0.52\textwidth}
      The \textbf{anti-Stokes field amplitude} generated by a
      degenerate pump probe reading out the Raman coherence:

      \vspace{0.5em}
      \[
        \boxed{
          C_2(\nuAS, t) = \int C_1(\Omega, t)\cdot\Epump(\nuAS - \Omega,\, t)\,d\Omega
        }
      \]

      \vspace{0.5em}
      \begin{itemize}
        \item $\nuAS = \nu_p^{\rm center} + \Omega$: anti-Stokes frequency
        \item This is $\omega_{\rm AS} = 2\omega_p - \omega_S$ at field level
        \item \textbf{Output center:} $\nuAS \approx 15\,292\wn \approx \SI{654}{nm}$
        \item The probe pump photon carries its own chirp $\Rightarrow$
              broadens the final linewidth
      \end{itemize}

    \column{0.45\textwidth}
      \begin{block}{Physical interpretation}
        The Raman coherence $C_1(\Omega, t)$ is
        "read out" by a second chirped pump photon.

        Because this probe is \emph{also} frequency-swept,
        it adds spectral uncertainty to the final CARS signal:
        \[
          \Delta\nuAS > \Delta\nu_{C_1}
        \]
      \end{block}

      \vspace{0.4em}
      \begin{block}{Implementation}
        Per-time-step FFT convolution\\
        (\texttt{fftconvolve})\\
        $\sim\SI{0.1}{s}$ for 1000 time steps
      \end{block}
  \end{columns}
\end{frame}

% ── Slide 23: Resolution–Signal Trade-off ────────────────────
\begin{frame}{The Resolution–Signal Trade-off}
  \begin{columns}[T]
    \column{0.5\textwidth}
      \textbf{More glass $\Rightarrow$ better resolution:}
      \begin{itemize}
        \item Larger GD spread $\Rightarrow$ longer pulse $\Rightarrow$\\
              slower chirp rate (ps/nm)
        \item At any instant, a narrower spectral\\
              slice is active
        \item $C_1$ RMS: $27\wn$ (A) vs.\ $59\wn$ (B)\\
              — factor $\mathbf{2.2\times}$ improvement
        \item $C_2$ RMS: $41\wn$ vs.\ $109\wn$\\
              — factor $\mathbf{2.6\times}$ improvement
      \end{itemize}

      \vspace{0.6em}
      \textbf{More glass $\Rightarrow$ lower peak signal:}
      \begin{itemize}
        \item Pump peak: $0.080\times$ (A) vs.\ $0.22\times$ (B)
        \item Longer pulse → lower instantaneous power
        \item CARS scales as $I_p^2 I_S$ → signal drops fast
      \end{itemize}

    \column{0.47\textwidth}
      \begin{block}{Design rule}
        For a target Raman linewidth $\Gamma$:
        \[
          d_{\rm glass} \propto \frac{\Delta\nu_{\rm TL}}{\Gamma}
        \]
        Increasing $d$ beyond what $\Gamma$ requires
        wastes peak power without gaining resolution.
      \end{block}

      \vspace{0.5em}
      \begin{alertblock}{Matched chirp rates are essential}
        Both beams must traverse the \emph{same} glass length
        so their chirp rates match.
        Mismatch (Scenario B) degrades resolution\\
        by factor $2\times$ or more.
      \end{alertblock}
  \end{columns}
\end{frame}

% ── Slide 24: Key Conclusions ────────────────────────────────
\begin{frame}{Key Conclusions}
  \begin{enumerate}
    \setlength{\itemsep}{0.55em}

    \item \textbf{S-TIH6 produces positive chirp.}
          Normal dispersion at 800–1050 nm: blue photons travel
          slower $\Rightarrow$ frequency sweeps red-to-blue at the sample.

    \item \textbf{Chirp elongates pulses dramatically.}
          Pump: $80\,\text{fs} \to 4.2\,\text{ps}$ ($52\times$);
          Stokes: $179\,\text{fs} \to 1.3\,\text{ps}$ ($7\times$) through 150 mm glass.
          Difference due to stronger dispersion at shorter wavelength.

    \item \textbf{Peak power drops but energy is conserved.}
          Pump peak ratio $= 0.080 = 1/\neff$ with $\neff \approx 12$.
          $\int I(t)\,dt$ ratio $= 1.000\;(\checkmark)$.

    \item \textbf{Matched chirp gives narrowest Raman linewidth.}
          Scenario A (15/15 cm): $C_1$ RMS $= 27.1\wn$, $C_2$ RMS $= 41.5\wn$. \\
          Mismatched pump (5 cm): $C_1 = 58.7\wn$, $C_2 = 108.5\wn$ — factor $2\times$ worse.

    \item \textbf{Degenerate probe broadens the anti-Stokes spectrum.}
          The second pump photon in $C_2$ adds spectral uncertainty:
          $\Delta\nu_{C_2} > \Delta\nu_{C_1}$ by $\sim\!50\,\%$.

  \end{enumerate}
\end{frame}

% ── Slide 25: Future Work ────────────────────────────────────
\begin{frame}{Limitations and Future Work}
  \begin{columns}[T]
    \column{0.5\textwidth}
      \textbf{Current limitations:}
      \begin{itemize}
        \item \textbf{GVD only} — group delay computed from first-order
              dispersion (group index). Third-order dispersion (TOD)
              and higher not yet included.
        \item \textbf{Transform-limited input} — assumes input pulses
              are perfectly unchirped. Real laser outputs have
              residual chirp.
        \item \textbf{Delay criterion} — optimal delay maximises
              signal, not resolution. For mismatched scenarios
              these optima differ.
        \item \textbf{1D spatial model} — no beam profile or
              focusing geometry.
      \end{itemize}

    \column{0.47\textwidth}
      \textbf{Next steps:}
      \begin{enumerate}
        \item Add \textbf{third-order dispersion} (TOD) to
              \texttt{build\_pulse} and verify pulse shape distortion
        \item Scan \textbf{glass thickness} continuously to produce
              a resolution vs.\ signal curve
        \item Verify that $C_1$ ridge is exactly horizontal
              for matched chirp rates (consistency check)
        \item Replace brute-force delay scan with
              \textbf{golden-section search}
        \item Extend to \textbf{experimental validation}:
              compare predicted $C_2$ linewidth with
              measured CARS spectra of lipid films
      \end{enumerate}
  \end{columns}
